use anyhow::Result;
use std::collections::HashMap;
use std::sync::Arc;

#[derive(Debug, thiserror::Error)]
pub enum ToolError {
    #[error("tool not found: {0}")]
    NotFound(String),
    #[error("tool execution failed: {0}")]
    ExecutionFailed(String),
}

/// A tool that can be called by an LLM.
pub trait Tool: Send + Sync {
    fn name(&self) -> &str;
    fn description(&self) -> &str;
    /// JSON Schema for the tool's parameters (OpenAI format)
    fn parameters(&self) -> serde_json::Value;
    /// Execute the tool. `arguments` is the JSON value from the LLM's tool call.
    fn execute(&self, arguments: &serde_json::Value) -> Result<String, ToolError>;
}

/// Registry of tools, keyed by name.
pub struct ToolSet {
    tools: HashMap<String, Arc<dyn Tool>>,
}

impl ToolSet {
    pub fn new() -> Self {
        Self {
            tools: HashMap::new(),
        }
    }

    /// Register a tool. Returns self for chaining.
    pub fn add(mut self, tool: impl Tool + 'static) -> Self {
        self.tools.insert(tool.name().to_string(), Arc::new(tool));
        self
    }

    /// Convert all registered tools to OpenAI-compatible tool definitions.
    pub fn to_openai_tools(&self) -> Vec<serde_json::Value> {
        self.tools
            .values()
            .map(|t| {
                serde_json::json!({
                    "type": "function",
                    "function": {
                        "name": t.name(),
                        "description": t.description(),
                        "parameters": t.parameters(),
                    }
                })
            })
            .collect()
    }

    /// Look up and execute a tool by name.
    pub fn call(&self, name: &str, arguments: &serde_json::Value) -> Result<String, ToolError> {
        let tool = self
            .tools
            .get(name)
            .ok_or_else(|| ToolError::NotFound(name.to_string()))?;
        tool.execute(arguments)
    }

    pub fn is_empty(&self) -> bool {
        self.tools.is_empty()
    }
}
```

use anyhow::Result;
use std::sync::Arc;

use super::openrouter::LlmClient;
use super::tool::{ToolError, ToolSet};

const MAX_ROUNDS: usize = 5;

#[derive(Debug)]
pub struct AgentResult {
    pub final_text: String,
    pub rounds: usize,
}

/// Run a multi-turn conversation where the LLM can call tools.
/// Each round: send messages + tools → if tool_calls, execute and append → repeat.
pub async fn agent_loop(
    llm: &LlmClient,
    tools: &Arc<ToolSet>,
    system_prompt: &str,
    user_message: &str,
) -> Result<AgentResult> {
    let tool_schemas = tools.to_openai_tools();
    let mut messages: Vec<serde_json::Value> = vec![
        serde_json::json!({"role": "system", "content": system_prompt}),
        serde_json::json!({"role": "user", "content": user_message}),
    ];

    for round in 0..MAX_ROUNDS {
        let body = serde_json::json!({
            "model": llm.model,
            "messages": messages,
            "max_tokens": 4096,
            "temperature": 0.7,
            "tools": tool_schemas,
        });

        let resp = llm
            .chat_with_tools(&body.to_string())
            .await?;

        let choice = match resp.choices.into_iter().next() {
            Some(c) => c,
            None => {
                return Ok(AgentResult {
                    final_text: String::new(),
                    rounds: round + 1,
                });
            }
        };

        // If LLM returned text (no tool calls), we're done
        if let Some(ref content) = choice.message.content {
            if !content.is_empty() {
                return Ok(AgentResult {
                    final_text: content.clone(),
                    rounds: round + 1,
                });
            }
        }

        // If LLM requested tool calls, execute them
        if let Some(ref tc_list) = choice.message.tool_calls {
            // Append the assistant's tool call message
            messages.push(serde_json::json!({
                "role": "assistant",
                "tool_calls": tc_list,
            }));

            // Execute each tool call and append results
            for tc in tc_list {
                let fn_name = tc.function.name.as_str();
                let fn_args: serde_json::Value = serde_json::from_str(
                    &tc.function.arguments,
                )
                .unwrap_or(serde_json::Value::Null);

                let result = match tools.call(fn_name, &fn_args) {
                    Ok(output) => output,
                    Err(e) => format!("Error: {e}"),
                };

                messages.push(serde_json::json!({
                    "role": "tool",
                    "tool_call_id": tc.id,
                    "content": result,
                }));
            }
            // Continue to next round
            continue;
        }

        // Neither text nor tool calls — shouldn't happen, but break out
        return Ok(AgentResult {
            final_text: String::new(),
            rounds: round + 1,
        });
    }

    Ok(AgentResult {
        final_text: "已达到最大轮次，分析未完成。".to_string(),
        rounds: MAX_ROUNDS,
    })
}

/// Non-streaming chat with tool support. Returns raw ToolChatResponse for agent loop.
pub async fn chat_with_tools(&self, body_json: &str) -> Result<super::agent::ToolChatResponse> {
    let resp = self
        .client
        .post(&self.api_endpoint)
        .bearer_auth(&self.api_key)
        .header("Content-Type", "application/json")
        .body(body_json.to_string())
        .send()
        .await?;

    let status = resp.status();
    if !status.is_success() {
        let body = resp.text().await.unwrap_or_default();
        anyhow::bail!("LLM API error {status}: {body}");
    }
    let response: super::agent::ToolChatResponse = resp.json().await?;
    Ok(response)
}
```
