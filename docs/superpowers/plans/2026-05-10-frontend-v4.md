- [ ] **Step 2: Add computed values for new cards**

```typescript
// New props: promotionRate, promotionHistoryAvg, sentimentTemp, netFlow, flowSector
const props = defineProps<{
  overview: { rise_count: number; fall_count: number; limit_up_count: number; limit_down_count: number; bomb_rate: number }
  riskLevel: string
  decisionSummary: string
  // New (optional — use defaults when backend not ready)
  promotionRate?: number
  promotionHistoryAvg?: number
  sentimentTemp?: number
  netFlow?: number
  flowSector?: string
}>()

const sentimentLabel = computed(() => {
  const t = props.sentimentTemp
  if (!t) return '暂无数据'
  if (t >= 75) return '偏热'
  if (t >= 50) return '正常'
  return '偏冷'
})

const sentimentColorClass = computed(() => {
  const t = props.sentimentTemp
  if (!t) return 'text-ink-muted'
  if (t >= 75) return 'text-up'
  if (t >= 50) return 'text-ai'
  return 'text-down'
})

const sentimentBorderClass = computed(() => {
  const t = props.sentimentTemp
  if (!t) return 'border-l-ink-dim'
  if (t >= 75) return 'border-l-up'
  if (t >= 50) return 'border-l-ai'
  return 'border-l-down'
})

const bombLabel = computed(() => {
  const r = props.overview.bomb_rate
  if (r > 0.3) return '偏高 注意'
  if (r > 0.2) return '正常偏高'
  return '偏低 惜售'
})

const promoColorClass = computed(() => {
  const r = props.promotionRate
  if (!r) return 'text-ink-muted'
  if (r >= 0.3) return 'text-up'
  return 'text-down'
})

const promoLabel = computed(() => {
  if (!props.promotionHistoryAvg) return ''
  const r = props.promotionRate
  const avg = props.promotionHistoryAvg
  if (!r || !avg) return ''
  return r >= avg ? '↑ 历史均' : '↓ 历史均'
})

const flowDisplay = computed(() => {
  const f = props.netFlow
  if (f == null) return '--'
  const abs = Math.abs(f)
  const unit = abs >= 1e8 ? '亿' : '万'
  const val = abs >= 1e8 ? (abs / 1e8).toFixed(1) : (abs / 1e4).toFixed(0)
  return (f >= 0 ? '+' : '-') + val + unit
})

const flowColorClass = computed(() => {
  const f = props.netFlow
  if (f == null) return 'text-ink-muted'
  return f >= 0 ? 'text-up' : 'text-down'
})

const flowSector = computed(() => props.flowSector || '--')
```

<template>

    <h3 class="text-caption-strong text-ink-muted mb-4 uppercase tracking-wider">情绪历史走势（近10日）</h3>
    <div ref="chartContainer" style="height: 200px"></div>
    <p class="text-caption text-ink-muted mt-3">{{ interpretation }}</p>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted, watch } from 'vue'
import { createChart, LineSeries, type IChartApi, type ISeriesApi, type Time } from 'lightweight-charts'

const props = defineProps<{
  data?: { date: string; value: number }[]
}>()

const chartContainer = ref<HTMLElement>()
let chart: IChartApi | null = null
let lineSeries: ISeriesApi<'Line'> | null = null

const interpretation = ref('')

function renderChart() {
  if (!chartContainer.value || !props.data?.length) return

  chart = createChart(chartContainer.value, {
    width: chartContainer.value.clientWidth,
    height: 200,
    layout: { background: { color: '#111124' }, textColor: '#888780', fontSize: 10 },
    grid: { vertLines: { color: 'rgba(255,255,255,0.06)' }, horzLines: { color: 'rgba(255,255,255,0.06)' } },
    timeScale: { borderColor: 'rgba(255,255,255,0.06)', timeVisible: false },
    rightPriceScale: { borderColor: 'rgba(255,255,255,0.06)' },
  })

  lineSeries = chart.addSeries(LineSeries, {
    color: '#7F77DD',
    lineWidth: 2,
    crosshairMarkerVisible: true,
    crosshairMarkerRadius: 4,
    lastValueVisible: true,
    priceLineVisible: false,
  })

  const chartData = props.data.map((d, i) => ({ time: i as Time, value: d.value }))
  lineSeries.setData(chartData)

  // Mark today's value
  const last = props.data[props.data.length - 1]
  if (last) {
    lineSeries.createPriceLine({
      price: last.value,
      color: '#7F77DD',

      title: `今日 ${last.value}`,
    })
  }

  // Interpretation
  const recent = props.data.slice(-3)
  if (recent.every(d => d.value >= 75)) {
    interpretation.value = '连续3日偏热区间，历史规律显示此后2~3日有回调风险，注意控仓'
  } else if (recent.every(d => d.value >= 50)) {
    interpretation.value = '情绪正常偏暖，短线操作环境尚可'
  } else {
    interpretation.value = '情绪偏冷，等待回暖信号'
  }

  chart.timeScale().fitContent()
}

onUnmounted(() => { chart?.remove(); chart = null })

const modes = [
  { id: 'pre' as const, label: '盘前' },
  { id: 'live' as const, label: '盘中' },
  { id: 'review' as const, label: '复盘' },
]

// Auto-detect mode based on time
function detectMode() {
  const h = new Date().getHours()
  const m = new Date().getMinutes()
  const mins = h * 60 + m
  if (mins >= 9 * 60 && mins < 15 * 60 + 10) activeMode.value = 'live'
  else if (mins >= 15 * 60 + 30) activeMode.value = 'review'
  else activeMode.value = 'pre'
}
onMounted(detectMode)
